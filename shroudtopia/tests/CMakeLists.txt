cmake_minimum_required(VERSION 3.20)
project(shroudtopia_loader_tests LANGUAGES C CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
add_library(xhl_fixture SHARED fixture.cpp)
add_library(flame_fixture SHARED fixture.cpp)
target_compile_definitions(flame_fixture PRIVATE FLAME_MOD)
add_library(valid_fixture SHARED fixture.cpp)
target_compile_definitions(valid_fixture PRIVATE VALID_MOD)
add_library(null_fixture SHARED fixture.cpp)
target_compile_definitions(null_fixture PRIVATE VALID_MOD NULL_FACTORY)
add_executable(loader_tests loader_tests.cpp)
foreach(target xhl_fixture flame_fixture valid_fixture null_fixture loader_tests)
  target_include_directories(${target} PRIVATE .. ../../include ../../third-party/nlohmann-json/include)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
endforeach()
add_dependencies(loader_tests xhl_fixture flame_fixture valid_fixture null_fixture)
enable_testing()
add_test(NAME shroudtopia_loader COMMAND loader_tests)

set(MINHOOK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third-party/minhook-1.3.4")
file(READ "${MINHOOK_ROOT}/SHROUDWEAVE-PROVENANCE.json" MINHOOK_PROVENANCE)
string(JSON MINHOOK_INPUT_COUNT LENGTH "${MINHOOK_PROVENANCE}" files)
math(EXPR MINHOOK_INPUT_LAST "${MINHOOK_INPUT_COUNT}-1")
foreach(index RANGE 0 ${MINHOOK_INPUT_LAST})
 string(JSON relative MEMBER "${MINHOOK_PROVENANCE}" files ${index})
 string(JSON expected GET "${MINHOOK_PROVENANCE}" files "${relative}")
 file(SHA256 "${MINHOOK_ROOT}/${relative}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Pinned repaired MinHook input changed: ${relative}")
 endif()
endforeach()
add_library(shroudweave_minhook STATIC
 "${MINHOOK_ROOT}/src/buffer.c" "${MINHOOK_ROOT}/src/hook.c"
 "${MINHOOK_ROOT}/src/trampoline.c" "${MINHOOK_ROOT}/src/hde/hde64.c")
target_include_directories(shroudweave_minhook PUBLIC "${MINHOOK_ROOT}/include")
add_executable(detour_service_tests detour_service_tests.cpp detour_fixture.asm ../process_detour_owner.cpp)
target_include_directories(detour_service_tests PRIVATE ..)
target_link_libraries(detour_service_tests PRIVATE shroudweave_minhook)
add_executable(detour_failure_tests detour_failure_tests.cpp detour_fixture.asm ../process_detour_owner.cpp)
target_include_directories(detour_failure_tests PRIVATE .. "${MINHOOK_ROOT}/include")
foreach(mode success invalid foreign concurrent)
 add_test(NAME detour_service_${mode} COMMAND detour_service_tests ${mode})
endforeach()
add_test(NAME detour_failed_enable COMMAND detour_failure_tests)
add_library(detour_lifetime_fixture SHARED detour_lifetime_fixture.cpp detour_fixture.asm)
add_test(NAME detour_image_residency COMMAND detour_service_tests residency $<TARGET_FILE:detour_lifetime_fixture>)
