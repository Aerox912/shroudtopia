cmake_minimum_required(VERSION 3.20)
project(shroudtopia_loader_tests LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
add_library(xhl_fixture SHARED fixture.cpp)
add_library(valid_fixture SHARED fixture.cpp)
target_compile_definitions(valid_fixture PRIVATE VALID_MOD)
add_library(null_fixture SHARED fixture.cpp)
target_compile_definitions(null_fixture PRIVATE VALID_MOD NULL_FACTORY)
add_executable(loader_tests loader_tests.cpp)
foreach(target xhl_fixture valid_fixture null_fixture loader_tests)
  target_include_directories(${target} PRIVATE .. ../../include ../../third-party/nlohmann-json/include)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
endforeach()
add_dependencies(loader_tests xhl_fixture valid_fixture null_fixture)
enable_testing()
add_test(NAME shroudtopia_loader COMMAND loader_tests)
