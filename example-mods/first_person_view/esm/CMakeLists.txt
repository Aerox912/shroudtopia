cmake_minimum_required(VERSION 3.24)
project(first_person_esm_scaffold LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The ESM checkpoint requires Windows x64")
endif()
set(ESM_FIXED_SDK "" CACHE PATH "Extracted accepted fixed-v1 SDK checkpoint")
foreach(header IN ITEMS runtime module)
  file(SHA256 "${ESM_FIXED_SDK}/include/esm/${header}_v1.h" actual)
  if(header STREQUAL "runtime")
    set(expected d74621d51bb475c4dd422de63aeb0efc1b4ce6fe8dd063e107853f93ea2cb65a)
  else()
    set(expected 840e1b02d34a302f7d58f876b655a02c5f252a040ccf7b9d10b1721dd0c17b9a)
  endif()
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "${header}_v1.h differs from the accepted checkpoint")
  endif()
endforeach()
add_library(first_person_esm_scaffold SHARED module.cpp)
target_include_directories(first_person_esm_scaffold PRIVATE "${ESM_FIXED_SDK}/include")
target_compile_features(first_person_esm_scaffold PRIVATE cxx_std_17)
add_executable(first_person_esm_boundary_test boundary_test.cpp)
target_include_directories(first_person_esm_boundary_test PRIVATE "${ESM_FIXED_SDK}/include")
target_compile_features(first_person_esm_boundary_test PRIVATE cxx_std_17)
enable_testing()
add_test(NAME first_person_esm_boundary COMMAND first_person_esm_boundary_test "$<TARGET_FILE:first_person_esm_scaffold>")
add_executable(first_person_settings_test settings_test.cpp)
target_compile_features(first_person_settings_test PRIVATE cxx_std_17)
add_test(NAME first_person_settings COMMAND first_person_settings_test)
add_executable(first_person_view_mode_test view_mode_test.cpp)
target_compile_features(first_person_view_mode_test PRIVATE cxx_std_17)
add_test(NAME first_person_view_mode COMMAND first_person_view_mode_test)
